<Title>Component Usage React</Title>
<Subtitle>Use Parlamentsdienste components in controlled and uncontrolled React patterns.</Subtitle>

<Subheading>Controlled Input</Subheading>

<Source
    language="jsx"
    code={`
import { PdInput } from '@parlamentsdienste/pdcomponents-react';
import { useState } from 'react';

function ControlledInputDemo() {
const [inputValue, setInputValue] = useState('Hello World');
return <PdInput value={inputValue} placeholder="Type something..." onInput={e => setInputValue(e.target.value)} />;
}
`}
/>

<Subheading>Uncontrolled Input with Ref</Subheading>

<Source
    language="jsx"
    code={`
import { PdInput } from '@parlamentsdienste/pdcomponents-react';
import { useRef } from 'react';

function UncontrolledInputDemo() {
const inputRef = useRef(null);
return (
<>
<PdInput placeholder="Type something..." value="test" ref={inputRef} />
<button
onClick={() => {
alert(inputRef.current?.value);
}}>
Show Value
</button>
</>
);
}
`}
/>
