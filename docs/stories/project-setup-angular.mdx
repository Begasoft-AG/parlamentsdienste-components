import { Meta, Source, Subheading, Subtitle, Title } from '@storybook/addon-docs/blocks';

<Meta title="Docs/Angular/Angular Project Setup" />

<Title>Project Setup Angular</Title>
<Subtitle>Install and configure the Angular package.</Subtitle>

<Subheading>Start a new Angular project using the Angular CLI</Subheading>

<Source language="bash" code={`ng new my-angular-app`} />

<Subheading>Install the parlamentsdienste components from npm</Subheading>

<Source
    language="bash"
    code={`cd my-angular-app
npm install @parlamentsdienste/pdcomponents-angular`}
/>

<Subheading>Add fonts and icons in angular.json assets</Subheading>
<p>
    Add this object to the <code>angular.json</code> <code>assets</code> array:
</p>

<Source
    language="json"
    code={`
    "assets": [
          {
            "glob": "**/*",
            "input": "node_modules/@parlamentsdienste/pdcomponents-core/dist/parlamentsdienstecore/assets",
            "output": "assets"
          },
    ]
`}
/>

<Subheading>Add base styles and fonts in angular.json styles</Subheading>
<p>
    Add these two lines to the <code>styles</code> array in <code>angular.json</code>:
</p>

<Source
    language="json"
    code={`
    "styles": [
        "node_modules/@parlamentsdienste/pdcomponents-core/dist/parlamentsdienstecore/typography.css",
        "node_modules/@parlamentsdienste/pdcomponents-core/dist/parlamentsdienstecore/parlamentsdienstecore.css"
    ],
`}
/>

<Subheading>Set the asset path in your app initializer</Subheading>
<p>
    Call <code>setAssetPath</code> and set it to the base URL of your project:
</p>

<Source
    language="typescript"
    code={`
import { setAssetPath } from '@parlamentsdienste/pdcomponents-core/components';

bootstrapApplication(AppComponent, {
providers: [
provideAppInitializer(() => {
setAssetPath('http://localhost:4200');
}),
provideZoneChangeDetection({ eventCoalescing: true }),
provideRouter(appRoutes),
],
}).catch(err => console.error(err));
`}
/>
