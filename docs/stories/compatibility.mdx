export const parseVersion = value =>
    String(value)
        .split('.')
        .map(part => Number.parseInt(part, 10));

export const compareVersion = (left, right) => {
    const leftParts = parseVersion(left);
    const rightParts = parseVersion(right);
    const maxLength = Math.max(leftParts.length, rightParts.length);

    for (let index = 0; index < maxLength; index += 1) {
        const leftPart = leftParts[index] ?? 0;
        const rightPart = rightParts[index] ?? 0;

        if (leftPart !== rightPart) {
            return leftPart - rightPart;
        }
    }

    return 0;

};

export const sortRows = rows =>
    [...rows].sort((left, right) => {
        const releaseComparison = compareVersion(right.compLibVersion, left.compLibVersion);

        if (releaseComparison !== 0) {
            return releaseComparison;
        }

        return compareVersion(right.frameworkVersion, left.frameworkVersion);
    });

export function StatusBadge({ status }) {
    const config = statusConfig[status];

    return (
        <span className={`compat-status compat-status-${status}`} style={{ color: config.color }}>
            <pd-icon name={config.icon} size="2.5" style={{ fill: config.color }}></pd-icon>
            <span>{config.label}</span>
        </span>
    );

}

export function CompatibilityTable({ columns, rows }) {
    return (
        <table className="compat-table">
            <thead>
                <tr>
                    {columns.map(column => (
                        <th key={column.key} scope="col">
                            {column.label}
                        </th>
                    ))}
                </tr>
            </thead>
            <tbody>
                {sortRows(rows).map((row, index) => (
                    <tr key={`${row.compLibVersion}-${row.frameworkVersion}-${row.nodeVersion}-${index}`}>
                        {columns.map(column => (
                            <td key={column.key}>
                                {column.key === 'status' ? <StatusBadge status={row.status} /> : row[column.key] || '-'}
                            </td>
                        ))}
                    </tr>
                ))}
            </tbody>
        </table>
    );
}
