`}

<Subheading>Two-way binding</Subheading>

<Source
    language="typescript"
    code={`
import { PdInput } from '@parlamentsdienste/pdcomponents-vue';
import { defineComponent, ref } from 'vue';
export default defineComponent({
    name: 'MyComponent',
    components: { PdInput },
    setup() {
        const twoWayBinding = ref('two-way binding');
        return { twoWayBinding };
    },
});
<template>
    <PdInput v-model="twoWayBinding" />
</template>;
`}
/>
